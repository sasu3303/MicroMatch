cmake_minimum_required(VERSION 3.20)
project(MicroMatch VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(ENABLE_ASAN "Address and undefined-behavior sanitizers" OFF)
option(ENABLE_TSAN "Thread sanitizer (separate build)" OFF)
if(ENABLE_ASAN AND ENABLE_TSAN)
  message(FATAL_ERROR "Use separate ASan and TSan builds")
endif()
if(MSVC)
  add_compile_options(/W4 /permissive-)
else()
  add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  if(ENABLE_ASAN)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
  elseif(ENABLE_TSAN)
    add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
    add_link_options(-fsanitize=thread)
  endif()
endif()
find_package(Threads REQUIRED)
add_library(micromatch_core src/book.cpp src/protocol.cpp)
target_include_directories(micromatch_core PUBLIC include)
target_link_libraries(micromatch_core PUBLIC Threads::Threads)
foreach(target IN ITEMS micromatch micromatch_tests micromatch_bench)
  if(target STREQUAL "micromatch")
    set(source src/main.cpp)
  elseif(target STREQUAL "micromatch_tests")
    set(source tests/tests.cpp)
  else()
    set(source src/benchmark.cpp)
  endif()
  add_executable(${target} ${source})
  target_link_libraries(${target} PRIVATE micromatch_core)
endforeach()
if(UNIX)
  add_executable(micromatch_server src/server.cpp)
  target_link_libraries(micromatch_server PRIVATE micromatch_core)
endif()
add_executable(micromatch_cancel_bench src/cancel_benchmark.cpp)
target_link_libraries(micromatch_cancel_bench PRIVATE micromatch_core)
enable_testing()
add_test(NAME matching_and_concurrency COMMAND micromatch_tests)
set_tests_properties(matching_and_concurrency PROPERTIES TIMEOUT 45)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(UNIX AND Python3_FOUND)
  add_test(NAME tcp_protocol COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_tcp.py $<TARGET_FILE:micromatch_server>)
  set_tests_properties(tcp_protocol PROPERTIES TIMEOUT 30)
endif()
